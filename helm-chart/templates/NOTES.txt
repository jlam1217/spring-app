    kubectl get -n {{ .Release.Namespace }} pods,svc,deploy

    kubectl -n {{ .Release.Namespace }} get svc {{ include "my-spring-app.fullname" . }}

To uninstall:

    helm uninstall {{ .Release.Name }}
